            $accountno = $clbookkey['accountno'];
            $value_date = $clbookkey['value_date'];
            $bookdate = $value_date;
            $transfered_amount = $clbookkey['total_amount'] ?? 0;
            $book_amount = $clbookkey['total_amount'] ?? 0;
            $bankdescription = $clbookkey['description_bank'] ?? '';
            $transactionno = $clbookkey['chequeno'] ?? '';
            $transaction_date = $value_date;

                    $purposecategory = $pc[0]['pcategory'];
                    $categoryid = $pc[0]['pcategorycode'];
                    $sub_categoryid = $pc[0]['psub_categoryid'];
                    $sub_category = $pc[0]['psub_category'];
                }

                // Payee details
                $paytoname = $beneficiaryname = $paytoaccountno = $paytobankname = $paytoifsc = $paytoaccountid = "";
                $is_intermediate_transfer = "";

                if (in_array(intval($categoryid), [2, 3, 5, 6, 7, 8]) && !empty($paytoid)) {
                    $acqry = "SELECT * FROM tbl_master_account 
                              WHERE cmp_id = '$companyid' 
                              AND subcmp_id IN ('" . str_replace(",", "','", $subcompanyid) . "') 
                              AND projectid IN ('" . str_replace(",", "','", $projectid) . "') 
                              AND paytoid = '$paytoid'";

                    if ($acres->num_rows > 0) {
                        $pkey = $acres->fetch_assoc();
                        $paytoname = $pkey['accountname'];
                        $beneficiaryname = $pkey['accountname'];
                        $paytoaccountno = $pkey['accountno'];
                        $paytobankname = $pkey['bankname'];
                        $paytoifsc = $pkey['ifsccode'];
                        $paytoaccountid = $pkey['accountid'];
                        $is_intermediate_transfer = "yes";
                    } else {
                        $freturns[] = ['status' => 'Error', 'message' => 'Payee details not found in master table.', 'sno' => $banksno];
                        continue;
                    }
                } else {
                    $prpin = getpaytoinfo($mconn, $companyid, $subcompanyid, 'paytoid', $paytoid);
                    if (!empty($prpin)) {
                        $paytoname = $prpin[0]['paytoname'];
                        $beneficiaryname = $prpin[0]['paytoname'];
                        $paytoaccountno = $prpin[0]['accountno'];
                        $paytobankname = $prpin[0]['bankname'];
                        $paytoifsc = $prpin[0]['ifsccode'];
                    }
                }

                if (!empty($paytoid) && !empty($purposecode)) {
                    $finyeartype = getprojectdetails($mconn, $companyid, $subcompanyid, "projectid", $projectid)[0]['finyr'];
                    $financialYear = getFinancialYear($value_date, $finyeartype);
                    $tsno = GetMaxRefTempSno($mconn);
                    $newreference_no = getnewreferenceno($mconn, $companyid, $subcompanyid, $projectid, $accountno);

                    $mconn->begin_transaction();
                    try {
                        $mconn->query("INSERT INTO tbl_maintransaction_breakdown (companyid, subcompanyid, projectid, reference_no, beneficiary_name, amount, 
			                        purpose_categoryid, purpose_category, purpose_subcategoryid, purpose_subcategory, purpose, purpose_code, description, description_bank) VALUES ('$companyid', '$subcompanyid', '$projectid', '$newreference_no', '$beneficiaryname', '$transfered_amount', '$categoryid', '$purposecategory', '$sub_categoryid', '$sub_category','$purpose', '$purposecode', '$textremark', '$bankdescription')");

                        $mconn->query("INSERT INTO tbl_maintransaction (tsno, companyid, subcompanyid, projectid, subprojectid, finyear, request_no, reference_no, approvedby, bookdate, book_amount, transactionno, transaction_date, transaction_type, transfered_amount, beforeedit_amount, payto, paytoid, paytoaccountno, paytobankname, paytoifsc, paytoaccountid, is_intermediate_transfer, payfrombankname, payfromaccountno, payfromaccountid, payfrombankid, paymenttype, beneficierie, reconciliation, reconciliation_banksno, reconcilation_process, value_date, book_source, book_status, requesteddate, requestedby) VALUES ('$tsno', '$companyid', '$subcompanyid', '$projectid', '$subprojectid', '$financialYear', '$request_no', '$newreference_no', '$approvedby', '$bookdate', '$book_amount', '$transactionno', '$transaction_date', '$transaction_type', '$transfered_amount', '$transfered_amount', '$paytoname', '$paytoid', '$paytoaccountno', '$paytobankname', '$paytoifsc', '$paytoaccountid', '$is_intermediate_transfer', '$bankname', '$accountno', '$accountid', '$bankid', '$paymenttype', '$beneficierie', '$reconciliation', '$reconciliation_banksno', '$reconcilation_process', '$value_date', '$book_source', '$book_status', '$requesteddate', '$requestedby')");
                        $mconn->query("UPDATE tbl_closetobook SET reconciliation='Done', book_status='completed', 
                            actule_referenceno='$newreference_no', reference_no='$newreference_no', 
                            reconstructstatus='automaticfromhistory-" . date('Y-m-d H:i:s') . "' WHERE sno='$banksno'");
                        $mconn->commit();
                        $freturns[] = ['status' => 'Success', 'sno' => $banksno, 'refno' => $newreference_no];
                    } catch (Exception $e) {
                        $mconn->rollback();
                        $freturns[] = ['status' => 'Error', 'sno' => $banksno, 'message' => $e->getMessage()];
                    }
                } else {
                    $freturns[] = ['status' => 'Error', 'sno' => $banksno, 'message' => 'Missing required data (paytoid or purposecode).'];
                }
            }
        }
    } else {
        $freturns[] = ['status' => 'Error', 'message' => 'No unreconstructed data found.'];
    }

    return $freturns;
}
